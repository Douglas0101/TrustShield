# ==============================================================================
# TrustShield — Guia de Comandos Docker & Make
#
# PRIORIZE O USO DOS COMANDOS `make`.
# Eles são atalhos mais simples e seguros para as operações do dia a dia.
# A seção de `docker compose` puro é mantida como referência avançada.
# ==============================================================================

# ==============================================================================
# Guia Rápido com MAKE (Método Recomendado)
# ==============================================================================
# Observação: Execute os comandos 'make' da raiz do projeto.

# ------------------------------------------------------------------------------
# 1) Ciclo Completo (do zero à API pronta)
# ------------------------------------------------------------------------------
# Limpa a stack, sobe os containers, executa o pipeline de dados/treino,
# promove o melhor modelo e reinicia a API.
make quick

# ------------------------------------------------------------------------------
# 2) Subir e Parar a Stack
# ------------------------------------------------------------------------------
# Sobe todos os serviços (API, Dashboard, MLflow, etc.) e aguarda a API
# ficar pronta para uso.
make up

# Para a stack e remove os containers.
make down

# Reinicia a stack (equivale a 'make down && make up').
make restart

# ------------------------------------------------------------------------------
# 3) Executar o Pipeline passo a passo
# ------------------------------------------------------------------------------
# Cada comando executa uma etapa dentro do container da API.
make data       # Gera o dataset inicial
make features   # Constrói as features para o modelo
make train      # Treina o modelo
make eval       # Avalia o modelo treinado
make optimize   # Otimiza os hiperparâmetros do modelo
make promote    # Promove o melhor modelo para produção
make reload     # Reinicia a API para carregar o novo modelo
make wait       # Aguarda a API ficar saudável após o reinício

# ------------------------------------------------------------------------------
# 4) Comandos Úteis
# ------------------------------------------------------------------------------
# Forçar o rebuild das imagens Docker sem usar o cache.
make rebuild

# Ver os logs de todos os containers em tempo real.
make logs

# Ver os logs apenas do container da API.
make api-logs

# Verificar o status dos containers.
make ps

# Acessar o shell dentro do container da API para depuração.
make api-shell

# Limpar completamente a stack, incluindo volumes Docker.
make clean

# Teste rápido para verificar se a API está online e pronta.
make smoke

# ==============================================================================
# Sequência Detalhada com DOCKER COMPOSE (Referência Avançada)
# ==============================================================================
# Nota: A porta da API no host agora é 8080 por padrão (definido no Makefile).
# Para usar outra porta, exporte a variável de ambiente:
# export API_HOST_PORT=8001

# 1) Subir a stack
# O comando 'make up' encapsula esta lógica.
DOCKER_BUILDKIT=1 COMPOSE_DOCKER_CLI_BUILD=1 \
docker compose -p trustshield -f docker/docker-compose.yml up -d --build

# 2) Aguardar a API ficar saudável
# O comando 'make wait' faz isso automaticamente.
echo "Aguardando a API ficar saudável..."
while [[ "$(curl -s -o /dev/null -w ''%{http_code}'' http://localhost:8080/healthz)" != "200" ]]; do
    printf '.'
    sleep 2
done
echo " API está pronta!"

# 3) Executar pipeline de dados e treino
# Use 'make data', 'make features', 'make train', etc.
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.data.make_dataset
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.features.build_features
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.models.train_fraud_model
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.models.evaluate_models
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.models.optimization

# 4) Promover o melhor modelo
# Use 'make promote'.
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -c "from pathlib import Path; import shutil,sys; p=Path('outputs/models'); c=sorted(p.glob('isolation_forest_optimized_*.joblib'), key=lambda x: x.stat().st_mtime, reverse=True); sys.exit(0) if not c else (shutil.copy2(c[0], p/'default_model.joblib') or print('Promovido:', c[0].name))"

# 5) Reiniciar a API para carregar o novo modelo
# Use 'make reload'.
docker compose -p trustshield -f docker/docker-compose.yml restart trustshield-api

# 6) Testar a predição (exemplo de payload)
# Note a porta :8080
curl -s -X POST http://localhost:8080/predict \
  -H "Content-Type: application/json" \
  -d '{"amount":250,"use_chip":"Chip","current_age":40,"retirement_age":65,
       "birth_year":1984,"gender":"M","latitude":-20.31,"longitude":-40.29,
       "yearly_income":75000,"total_debt":15000,"credit_score":720,
       "num_credit_cards":4,"transaction_hour":15,"day_of_week":3,
       "is_weekend":false,"is_night_transaction":false,"amount_vs_avg":2.5}' | jq .

# 7) Parar e limpar a stack
# Use 'make down' ou 'make clean'.
docker compose -p trustshield -f docker/docker-compose.yml down --remove-orphans
