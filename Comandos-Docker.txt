# ==============================================================================
# TrustShield — Guia de Comandos Docker & Make (v3.1 - Otimizado)
#
# PRIORIZE O USO DOS COMANDOS `make` (atalhos seguros/automatizados).
# A seção de `docker compose` puro fica como referência avançada.
# ==============================================================================

# ==============================================================================
# 0) Portas & URLs (padrões)
# ==============================================================================
# API.............: http://localhost:8000
# Dashboard.......: http://localhost:8501
# MLflow..........: http://localhost:5000
# MinIO...........: http://localhost:9000
# MinIO Console...: http://localhost:9001
# (Para alterar portas: export API_HOST_PORT=XXXX etc.)

# ==============================================================================
# 1) Guia rápido com MAKE (recomendado)
# ==============================================================================
# Execute sempre da raiz do projeto.

# ---- Ciclos completos --------------------------------------------------------
make quick       # Limpa ambiente e executa ciclo completo do zero (otimizado com Optuna)
make cycle       # Executa o pipeline completo: data -> features -> optimize -> promote -> ...

# ---- Subir/derrubar stack ----------------------------------------------------
make check       # Verifica pré-requisitos (docker/compose/arquivos)
make build       # Constrói imagens
make up          # Sobe os serviços e aguarda a API ficar healthy
make down        # Derruba os serviços (mantém volumes)
make restart     # Reinicia toda a stack
make rebuild     # Recria imagens sem cache e reinicia
make wait        # Aguarda a API ficar healthy
make status      # Mostra URLs úteis
make env         # Mostra variáveis de ambiente/portas

# ---- Observabilidade e acessos -----------------------------------------------
make logs        # Logs de todos os serviços
make api-logs    # Logs da API
make dash-logs   # Logs do dashboard
make ps          # Status/containers
make top         # Processos/recursos
make api-shell   # Shell no container da API
make dash-shell  # Shell no container do dashboard

# ---- Pipeline (dentro dos containers) ----------------------------------------
make data        # Ingestão/limpeza de dados (src.data.make_dataset)
make features    # Engenharia de features (src.features.build_features)
make optimize    # [Principal] Executa otimização de hiperparâmetros com Optuna
make promote     # Promove o melhor modelo do 'optimize' -> outputs/models/default_model.joblib
make eval        # Avaliação de modelos
make validate    # Checks de qualidade (dados/modelo)
make interpret   # Interpretação/explicabilidade
make reload      # Reinicia a API para carregar o novo modelo
make smoke       # Health-check /healthz e /status
make train       # [Alternativo] Treina uma suíte de modelos (não usado no 'cycle' padrão)

# ---- Local (sem Docker) ------------------------------------------------------
make venv             # Cria .venv
make deps             # Instala dependências na .venv
make optimize-local   # Otimização local com Optuna
make train-local      # Treino alternativo local
make validate-local   # Validação local
make interpret-local  # Interpretação local

# ---- Limpeza -----------------------------------------------------------------
make clean       # Remove containers + volumes nomeados do projeto
make cache-clean # Limpa caches locais do projeto

# ==============================================================================
# 2) Referência com DOCKER COMPOSE (avançado)
# ==============================================================================
# Use sempre `docker compose` (com espaço) — a v1 `docker-compose` é legada.

# Parar/remover serviços do projeto:
docker compose -f docker/docker-compose.yml -p trustshield down --remove-orphans

# Construir e subir em background (com build):
DOCKER_BUILDKIT=1 COMPOSE_DOCKER_CLI_BUILD=1 \
  docker compose -f docker/docker-compose.yml -p trustshield up -d --build

# Aguardar a API ficar saudável (equivalente ao make wait):
echo "Aguardando a API ficar saudável..."
for i in $(seq 1 60); do
  code=$(curl -s -o /dev/null -w '%{http_code}' http://localhost:8000/healthz)
  [ "$code" = "200" ] && break
  printf '.'; sleep 2
done
echo " API está pronta!"

# Executar etapas do pipeline dentro da API:
docker compose -f docker/docker-compose.yml -p trustshield exec trustshield-api \
  python -m src.data.make_dataset
docker compose -f docker/docker-compose.yml -p trustshield exec trustshield-api \
  python -m src.features.build_features
docker compose -f docker/docker-compose.yml -p trustshield exec trustshield-api \
  python -m src.models.optimization --data data/features/featured_dataset.parquet

# Promover o melhor modelo para produção (equivale a make promote):
docker compose -f docker/docker-compose.yml -p trustshield exec trustshield-api \
  python -c "from pathlib import Path; import shutil, sys; p_opt=Path('outputs/optimization/optuna'); p_out=Path('outputs/models'); models=sorted(p_opt.glob('best_model_*.joblib'), key=lambda x: x.stat().st_mtime, reverse=True); (models and (shutil.copy2(models[0], p_out / 'default_model.joblib'), print(f'Promovido: {models[0].name}'))) or sys.exit('Nenhum modelo otimizado encontrado.')"

# Reiniciar a API para carregar o novo modelo:
docker compose -f docker/docker-compose.yml -p trustshield restart trustshield-api

# Teste de predição (exemplo) — note a porta :8000

streamlit run src/dashboard/app.py
