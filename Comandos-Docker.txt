# ==============================================================================
# TrustShield — Sequência Básica com DOCKER COMPOSE (sem Make)
# limpeza → dados → features → treino → avaliação → otimização → promoção → validação
# ==============================================================================

# Observações importantes:
# - MinIO no host: porta padrão agora é 9900 (console 9901).
#   Se quiser outras portas, exporte:
#     export MINIO_HOST_PORT=9900
#     export MINIO_CONSOLE_HOST_PORT=9901
# - MLflow UI no host: porta padrão agora é 5500 (container continua 5000).
#   Para customizar:
#     export MLFLOW_HOST_PORT=5500
# - Todos os comandos assumem que você está na RAIZ do projeto.
# - O docker-compose já cria o bucket S3 "mlflow" via serviço minio-init.

# ==============================================================================
# 0) VALIDAÇÃO DO COMPOSE EFETIVO (opcional, mas recomendado)
# ------------------------------------------------------------------------------
docker compose -p trustshield -f docker/docker-compose.yml config >/dev/null && echo "Compose OK"

# ==============================================================================
# 1) LIMPEZA / RESET DA STACK
# ------------------------------------------------------------------------------
DOCKER_BUILDKIT=1 COMPOSE_DOCKER_CLI_BUILD=1 \
docker compose -p trustshield -f docker/docker-compose.yml down --remove-orphans

# (Opcional) Limpeza mais pesada de recursos “sobrando” no host (cuidado!)
# docker system prune -af --volumes

# ==============================================================================
# 2) SUBIR A STACK (build + up) — COM PORTAS PARAMETRIZADAS
# ------------------------------------------------------------------------------
# Subida padrão (MinIO: 9900/9901; MLflow UI: 5500)
DOCKER_BUILDKIT=1 COMPOSE_DOCKER_CLI_BUILD=1 \
docker compose -p trustshield -f docker/docker-compose.yml up -d --build

# Se quiser forçar outras portas no host:
# MINIO_HOST_PORT=9800 MINIO_CONSOLE_HOST_PORT=9801 MLFLOW_HOST_PORT=5600 \
# DOCKER_BUILDKIT=1 COMPOSE_DOCKER_CLI_BUILD=1 \
# docker compose -p trustshield -f docker/docker-compose.yml up -d --build

# Garantir que a imagem do MLflow está atualizada (3.x):
docker compose -p trustshield -f docker/docker-compose.yml pull mlflow

# Verifique o status:
docker compose -p trustshield -f docker/docker-compose.yml ps
docker compose -p trustshield -f docker/docker-compose.yml logs -f trustshield-api

# ==============================================================================
# 3) AGUARDAR “HEALTHY” DA API E TESTAR /healthz
# ------------------------------------------------------------------------------
CID=$(docker compose -p trustshield -f docker/docker-compose.yml ps -q trustshield-api)
for i in $(seq 1 60); do
  STATUS=$(docker inspect -f '{{.State.Health.Status}}' "$CID" 2>/dev/null || echo "unknown")
  echo "tentativa $i: $STATUS"
  [ "$STATUS" = "healthy" ] && break
  sleep 2
done
[ "$STATUS" = "healthy" ] || { echo "ERRO: API não ficou healthy a tempo."; exit 1; }
curl -fsS http://127.0.0.1:8000/healthz && echo " - /healthz OK"

# ==============================================================================
# 4) PIPELINE DE DADOS E FEATURES (rodando dentro do container da API)
# ------------------------------------------------------------------------------
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -m src.data.make_dataset

docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -m src.features.build_features

# ==============================================================================
# 5) TREINO, AVALIAÇÃO, OTIMIZAÇÃO (gera *_optimized_*.joblib)
# ------------------------------------------------------------------------------
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -m src.models.train_fraud_model

docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -m src.models.evaluate_models

docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -m src.models.optimization

# (Opcional) Validação e interpretabilidade
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -m src.models.validation

docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -m src.models.interpretation

# ==============================================================================
# 6) PROMOVER O MELHOR MODELO PARA default_model.joblib (SEM HEREDOC)
# ------------------------------------------------------------------------------
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -c "from pathlib import Path; import shutil; p=Path('outputs/models'); \
c=sorted(p.glob('isolation_forest_optimized_*.joblib'), key=lambda x: x.stat().st_mtime, reverse=True); \
print('Nenhum modelo otimizado encontrado.') if not c else (shutil.copy2(c[0], p/'default_model.joblib') or print(f'Promovido: {c[0].name}'))"

# ==============================================================================
# 7) REINICIAR A API E VALIDAR READINESS (/status)
# ------------------------------------------------------------------------------
docker compose -p trustshield -f docker/docker-compose.yml restart trustshield-api

CID=$(docker compose -p trustshield -f docker/docker-compose.yml ps -q trustshield-api)
for i in $(seq 1 60); do
  STATUS=$(docker inspect -f '{{.State.Health.Status}}' "$CID" 2>/dev/null || echo "unknown")
  echo "tentativa $i: $STATUS"
  [ "$STATUS" = "healthy" ] && break
  sleep 2
done
[ "$STATUS" = "healthy" ] || { echo "ERRO: API não ficou healthy após restart."; exit 1; }

curl -fsS http://127.0.0.1:8000/status && echo " - /status OK"

# ==============================================================================
# 8) TESTE RÁPIDO /predict (ajuste o payload ao seu schema)
# ------------------------------------------------------------------------------
curl -s -X POST http://127.0.0.1:8000/predict \
  -H "Content-Type: application/json" \
  -d '{"amount":250,"use_chip":"Chip","current_age":40,"retirement_age":65,
       "birth_year":1984,"gender":"M","latitude":-20.31,"longitude":-40.29,
       "yearly_income":75000,"total_debt":15000,"credit_score":720,
       "num_credit_cards":4,"transaction_hour":15,"day_of_week":3,
       "is_weekend":false,"is_night_transaction":false,"amount_vs_avg":2.5}' | jq .

# ==============================================================================
# 9) CICLO COMPLETO EM UMA LINHA (SEM HEREDOCS)
# ------------------------------------------------------------------------------
DOCKER_BUILDKIT=1 COMPOSE_DOCKER_CLI_BUILD=1 docker compose -p trustshield -f docker/docker-compose.yml down --remove-orphans && \
MINIO_HOST_PORT=${MINIO_HOST_PORT:-9900} MINIO_CONSOLE_HOST_PORT=${MINIO_CONSOLE_HOST_PORT:-9901} MLFLOW_HOST_PORT=${MLFLOW_HOST_PORT:-5500} \
DOCKER_BUILDKIT=1 COMPOSE_DOCKER_CLI_BUILD=1 docker compose -p trustshield -f docker/docker-compose.yml up -d --build && \
CID=$(docker compose -p trustshield -f docker/docker-compose.yml ps -q trustshield-api) && \
for i in $(seq 1 60); do STATUS=$(docker inspect -f '{{.State.Health.Status}}' "$CID" 2>/dev/null || echo "unknown"); \
echo "tentativa $i: $STATUS"; [ "$STATUS" = "healthy" ] && break; sleep 2; done && \
[ "$STATUS" = "healthy" ] && \
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.data.make_dataset && \
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.features.build_features && \
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.models.train_fraud_model && \
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.models.evaluate_models && \
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api python -m src.models.optimization && \
docker compose -p trustshield -f docker/docker-compose.yml exec trustshield-api \
  python -c "from pathlib import Path; import shutil; p=Path('outputs/models'); \
c=sorted(p.glob('isolation_forest_optimized_*.joblib'), key=lambda x: x.stat().st_mtime, reverse=True); \
assert c, 'Nenhum modelo otimizado encontrado.'; shutil.copy2(c[0], p/'default_model.joblib'); \
print(f'Promovido: {c[0].name}')" && \
docker compose -p trustshield -f docker/docker-compose.yml restart trustshield-api && \
CID=$(docker compose -p trustshield -f docker/docker-compose.yml ps -q trustshield-api) && \
for i in $(seq 1 60); do STATUS=$(docker inspect -f '{{.State.Health.Status}}' "$CID" 2>/dev/null || echo "unknown"); \
echo "tentativa $i: $STATUS"; [ "$STATUS" = "healthy" ] && break; sleep 2; done && \
curl -fsS http://127.0.0.1:8000/status && echo " - /status OK"
